Investigation of scp_wrapper.sh revealed a restricted shell environment that explicitly forces the use of scp .

┌──(kali㉿kali)-[~/Downloads/home/max]
└─$ cat scp_wrapper.sh      
#!/bin/bash
case $SSH_ORIGINAL_COMMAND in
 'scp'*)
    $SSH_ORIGINAL_COMMAND
    ;;
 *)
    echo "ACCESS DENIED."
    scp
    ;;
esac
